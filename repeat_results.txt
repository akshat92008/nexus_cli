const express = require('express');
const app = express();
const port = 3000;
const db = require('./db');

app.get('/', (req, res) => {
    try {
        const connectionStatus = db.getConnectionStatus();
        if (connectionStatus === 'connected') {
            return res.status(200).json({ status: 'healthy' });
        } else {
            return res.status(503).json({ status: 'degraded', message: 'Database connection failed' });
        }
    } catch (error) {
        console.error('Error checking database connection:', error);
        return res.status(200).json({ status: 'degraded', message: 'Database connection failed' });
    }
});

app.listen(port, () => {
  console.log(`Server is running on http://localhost:${port}`);
});

const app = express();
const port = 3000;

app.get('/healthcheck', (req, res) => {
    try {
        const connection = db.getConnection(); // Attempt to connect to the database
        if (!connection) {
            throw new Error('Database connection failed');
        }
        connection.end(); // Close the connection
        return res.status(200).json({ status: 'healthy' });
    } catch (error) {
        console.error('Healthcheck error:', error);
        return res.status(500).json({ status: 'degraded', message: 'Database connection failed' });
    }
});

app.listen(port, () => {
    console.log(`Server is running on http://localhost:${port}`);
});
